    <script>
      // Al cargar el documento, comprobamos si tenemos un usuario guardado previamente
      document.addEventListener("DOMContentLoaded", () => {
        const usuarioGuardado = localStorage.getItem("remembered_admin_user");
        if (usuarioGuardado) {
          document.getElementById("log-user").value = usuarioGuardado;
        }
      });

      // Función para mostrar la contraseña durante 5 segundos exactos
      function revelarContrasenaTemporaria() {
        const passInput = document.getElementById("log-pass");
        if (passInput.type === "password") {
          passInput.type = "text";
          clearTimeout(temporizadorContrasena); // Limpiamos cualquier temporizador previo
          temporizadorContrasena = setTimeout(() => {
            passInput.type = "password";
          }, 5000); // 5000ms = 5 segundos
        } else {
          passInput.type = "password"; // Si ya estaba visible, lo ocultamos inmediatamente
          clearTimeout(temporizadorContrasena);
        }
      }

      // Interceptor del Login
      document.getElementById("form-login").addEventListener("submit", async (e) => {
        e.preventDefault();
        const usuario = document.getElementById("log-user").value.trim();
        const pass = document.getElementById("log-pass").value;

        // CORRECCIÓN: Esperar un instante o re-inicializar localmente si script.js asincrónico no terminó
        if (!window.supabase && typeof supabasejs !== "undefined" && typeof SUPABASE_URL !== "undefined") {
          window.supabase = supabasejs.createClient(SUPABASE_URL, SUPABASE_KEY);
        }

        try {
            const activeSupabase = window.supabase || supabase;
            // Validamos que el cliente de Supabase esté listo antes de consultar
            if (!supabase) {
            alert("El servicio de conexión aún se está inicializando. Por favor, aguarde 2 segundos y vuelva a intentar.");
            return;
          }

          // Buscamos en la tabla de Supabase utilizando la instancia global calibrada
          const { data, error } = await supabase
            .from("administadores")
            .select("*")
            .or(`usuario.eq.${usuario},email.eq.${usuario}`);

          if (error) throw error;

          if (data && data.length > 0) {
            // Guardar el identificador en local storage para evitar tener que tipearlo de nuevo en el futuro
            localStorage.setItem("remembered_admin_user", usuario);

            // Transición visual limpia a los paneles operativos
            document.getElementById("login-section").classList.add("hidden");
            document.getElementById("dashboard-section").classList.remove("hidden");
            
            // Ejecutamos las llamadas con nombres reales de funciones presentes en tu script.js
            await cargarViajes();
            await cargarPresupuestosAdmin();
            if( typeof cargarGaleriaAdmin === "function") {
              await cargarGaleriaAdmin();
                } 
            } else {
            alert("Credenciales incorrectas o usuario no registrado.");
            }
        } catch (err) {
          console.error("Error crítico de enlace a Supabase:", err);
          alert("Error de conexión al validar accesos. Verifique que sus claves en .env.local sean válidas.");
        }
      });

      // Auxiliar por si en tu script.js la carga de viajes de admin se llama cargarViajes()
      async function cargarViajes() {
        if (typeof cargarViajes === "function") {
          await cargarViajes();
        } else if (typeof cargarViajesAdmin === "function") {
          await cargarViajesAdmin();
        }
      }

      function recuperarPassword() {
        const email = prompt("Introduce tu email registrado para enviarte las instrucciones:");
        if (email) {
          alert("Si el correo electrónico coincide con nuestra base de datos, se despachará un enlace de restablecimiento.");
        }
      }

      function cerrarSesion() {
        window.location.reload();
      }
